This page shows the code behind the [Train a Deep Learning Model](/docs/legacy/training) tutorial.

## The developers perspective

If you are a developer and you want to know how to integrate your own deep learning app into Arkitekt, you can read our Developer Tutorial on how to do this [here](/docs/develop/python).
But also please just find here **ALL** the code for the `Train CARE` and `Predict CARE` actions, which are just a few lines of code, and should be easy to understand.

The tutorial was recorded with the Paper version of Arkitekt, where these actions used
`from arkitekt import register` and `mikro.api.schema` fragments (`RepresentationFragment`,
`ContextFragment`, `ModelFragment`). The code below is the same pair of actions ported to the current
`arkitekt` and `mikro` packages, so the names you see in the tutorial's videos differ from the ones
here:

| Paper | Now |
|---|---|
| `@register()` | `app = App(...)` and `@app.action` |
| `RepresentationFragment` | `SingleChannelVolume`, a lens from `mikro.arkitekt.specs` |
| Training pairs found through a context and a `gt` link | two lists, `noisy` and `ground_truth`, in the same order |
| `ModelFragment` and `create_model(...)` | the zipped model stored as a mikro `File` with `mikro.from_file_like(...)` |
| `from_xarray(..., origins=[...])` | `mikro.create_array_dataset(...)` with a `derived_from` edge |
| `from arkitekt.tqdm import tqdm` | `task.progress(...)` on an injected `Task` |

For what each of these does, see [Core concepts](/docs/develop/python#core-concepts) and
[Types & data](/docs/develop/python/types).

```python
"""CARE denoising: train a model on noisy/clean pairs, then apply it."""

import shutil
import tempfile
from pathlib import Path
from typing import Annotated

import numpy as np
import xarray as xr
from csbdeep.data import RawData, create_patches
from csbdeep.io import load_training_data
from csbdeep.models import CARE, Config

from arkitekt import App, Description, Task, run
from mikro import Mikro, mikro_service
from mikro.api.schema import File
from mikro.arkitekt.specs import SingleChannelVolume

app = App("care", "0.1.0", services=[mikro_service])


def as_zyx(volume: SingleChannelVolume) -> xr.DataArray:
    """The stack behind the lens as a plain ZYX array (drops the size-1 channel and time axes)."""
    return volume.data.squeeze().transpose("z", "y", "x")


@app.action
def train_care_model(
    noisy: Annotated[list[SingleChannelVolume], Description("The noisy stacks")],
    ground_truth: Annotated[
        list[SingleChannelVolume], Description("The clean stacks, in the same order")
    ],
    epochs: Annotated[int, Description("Number of epochs")] = 100,
    patches_per_image: Annotated[int, Description("Patches cut from each stack")] = 1024,
    trainsteps_per_epoch: Annotated[int, Description("Training steps per epoch")] = 400,
    validation_split: Annotated[float, Description("Share of patches kept for validation")] = 0.1,
    *,
    mikro: Mikro,
    task: Task,
) -> File:
    """Train CARE Model

    Trains a CARE denoising model for {{epochs}} epochs on pairs of noisy and clean stacks,
    and stores the trained model as a file.
    """
    if len(noisy) != len(ground_truth):
        raise ValueError("Pass exactly one clean stack for every noisy stack")

    task.progress(5, f"Loading {len(noisy)} pairs")
    X = [np.asarray(as_zyx(v)) for v in noisy]
    Y = [np.asarray(as_zyx(v)) for v in ground_truth]

    with tempfile.TemporaryDirectory() as workdir:
        work = Path(workdir)

        # Everything from here to the upload is plain CSBDeep, see the CARE paper.
        create_patches(
            raw_data=RawData.from_arrays(X, Y, axes="ZYX"),
            patch_size=(16, 64, 64),
            n_patches_per_image=patches_per_image,
            save_file=str(work / "patches.npz"),
        )
        (X_train, Y_train), (X_val, Y_val), axes = load_training_data(
            str(work / "patches.npz"), validation_split=validation_split
        )
        config = Config(axes, train_steps_per_epoch=trainsteps_per_epoch)
        model = CARE(config, "care", basedir=str(work / "models"))

        # One epoch at a time, so the progress bar in Orkestrator moves with the training.
        for epoch in range(epochs):
            task.progress(10 + int(85 * epoch / epochs), f"Epoch {epoch + 1}/{epochs}")
            model.train(X_train, Y_train, validation_data=(X_val, Y_val), epochs=1)

        task.progress(95, "Uploading model")
        archive = shutil.make_archive(str(work / "care-model"), "zip", work / "models" / "care")
        return mikro.from_file_like(file=archive, file_name="care-model.zip")


@app.action
def predict_care(
    volume: Annotated[SingleChannelVolume, Description("The stack to denoise")],
    model: Annotated[File, Description("A model made by Train CARE Model")],
    *,
    mikro: Mikro,
    task: Task,
) -> SingleChannelVolume:
    """Predict CARE

    Denoises a stack with a trained CARE model and stores the result on the same grid.
    """
    source = as_zyx(volume)

    with tempfile.TemporaryDirectory() as workdir, model.as_path() as archive:
        task.progress(10, "Unpacking model")
        shutil.unpack_archive(archive, Path(workdir) / "care", format="zip")
        care = CARE(config=None, name="care", basedir=workdir)

        task.progress(30, "Predicting")
        restored = care.predict(np.asarray(source), "ZYX")

    task.progress(80, "Uploading")
    result = mikro.create_array_dataset(
        data=xr.DataArray(restored, dims=source.dims),
        scales=[],
        name="CARE denoised",
        axes=volume.carried_axes(source.dims),
        # Same grid as the input, values transformed.
        derived_from=[volume.derive_identity(value_relation="TRANSFORMED")],
    )
    return result.lens()


if __name__ == "__main__":
    run(app)
```
